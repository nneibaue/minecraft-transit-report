import subprocess
from bridge.settings import Settings
from deploy import deploy as d

root = s.server_dir / "world" / "computercraft" / "computer"
repo = {"chat.lua": "base/chat.lua", "client.lua": "turtle/client.lua"}
ours = b"-- startup.lua : turtle-helper"
tok = s.bridge_token.encode("utf-8")
def git_bytes(spec):
    return subprocess.run(["git", "show", spec], capture_output=True, check=True).stdout
def pre_push(rel):
    sha = subprocess.run(["git", "log", "-1", "--format=%H", "origin/main", "--", "turtle/turtle-helper/" + rel], capture_output=True, text=True, check=True).stdout.strip()
    return git_bytes("%s~1:turtle/turtle-helper/%s" % (sha, rel))
folders = [f for f in root.iterdir() if f.is_dir()]
marked = [f for f in folders if (f / "_marker.txt").is_file()]
installed = [f for f in folders if f not in marked and (f / "startup.lua").is_file() and (f / "startup.lua").read_bytes().startswith(ours)]
assert installed and len(marked) == 1, ([f.name for f in installed], [f.name for f in marked])
for f in installed:
    for name, rel in repo.items():
        now = (f / name).read_bytes()
        assert now == git_bytes("origin/main:turtle/turtle-helper/" + rel), (f.name, name, "update did not land")
        assert now != pre_push(rel), (f.name, name, "payload did not change this file")
    assert (f / "startup.lua").read_bytes() == git_bytes("origin/main:turtle/turtle-helper/startup.lua"), (f.name, "startup.lua")

    assert (f / "bridge.txt").read_bytes().strip() == b"ws://127.0.0.1:8765", (f.name, "bridge.txt changed")
